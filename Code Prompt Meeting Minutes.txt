import os

def generate_meeting_minutes(input_file_path, output_file_path, page_number=1):
    """
    อ่านไฟล์ transcript และสร้างรายงานสรุปการประชุมพร้อมระบุเลขหน้าในหัวตาราง
    """
    if not os.path.exists(input_file_path):
        print(f"Error: ไม่พบไฟล์ต้นฉบับที่ '{input_file_path}'")
        return

    # อ่านเนื้อหาจากไฟล์ Transcript
    with open(input_file_path, 'r', encoding='utf-8') as f:
        transcript_content = f.read()

    # จัดรูปแบบโครงสร้างสรุปการประชุม พร้อมใส่เลขหน้าในหัวตาราง
    minutes_template = f"""==================================================
MEETING MINUTES | PAGE: {page_number:02d}
==================================================
[ Metadata ]
- Document Type: International Public Health Collaboration Summary
- Status: Confirmed / Official Record

[ Meeting Details ]
- Date: [Insert Date]
- Time: [Insert Time]
- Venue / Platform: [Insert Virtual/Physical Location]
- Subject: Bilateral Collaboration & Public Health Policy Alignment

[ Attendees ]
- Agency A Representatives:
  * [Name 1], [Position/Title]
  * [Name 2], [Position/Title]
- Agency B / International Partners:
  * [Name 3], [Position/Title]
  * [Name 4], [Position/Title]

[ Agenda Items ]
1. Review of current bilateral health initiatives and data integration frameworks.
2. Discussion on policy alignment, data privacy (e.g., PDPA compliance), and cross-border data exchange.
3. Action plan review and milestone tracking for joint public health projects.

[ Discussions & Decisions ]
- Policy Alignment:
  * Emphasized the integration of health information systems to support vulnerable populations and elderly care frameworks.
  * Discussed protocols for secure data extraction, ensuring patient identifiers are masked in compliance with data privacy regulations.
- Technical & Operational Insights:
  * Reviewed interoperability standards between national health data platforms and regional digital health tools.
  * Decisions made to streamline reporting workflows and enhance multi-tier database architectures.

[ Action Items ]
- [ ] Action 1: Finalize the technical specifications for data integration protocols (Responsible: [Name], Deadline: [Date]).
- [ ] Action 2: Draft the revised bilateral memorandum incorporating data privacy guidelines (Responsible: [Name], Deadline: [Date]).
- [ ] Action 3: Schedule the next technical follow-up workshop (Responsible: [Name], Deadline: [Date]).

==================================================
TRANSCRIPT REFERENCE EXCERPT (Page {page_number}):
==================================================
{transcript_content[:1000]}...
[Note: Truncated for summary purposes]
"""

    # บันทึกผลลัพธ์ลงในไฟล์ .txt
    with open(output_file_path, 'w', encoding='utf-8') as f:
        f.write(minutes_template)

    print(f"สร้างรายงานสรุปการประชุม (หน้า {page_number}) สำเร็จ! บันทึกที่: {output_file_path}")

if __name__ == "__main__":
    # สามารถเปลี่ยนเลขหน้าตรงพารามิเตอร์นี้ได้เลยครับ (เช่น หน้า 1, หน้า 2)
    current_page = 1 
    
    input_transcript = "meeting_transcript.txt"
    output_minutes = f"public_health_meeting_minutes_page_{current_page}.txt"
    
    generate_meeting_minutes(input_transcript, output_minutes, page_number=current_page)
